  node scripts/import-teams.js

Option 4: TypeScript/Next.js Code

  const response = await fetch('/api/teams/add', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      name: 'Basketball Stars',
      photoUrl: 'https://example.com/logo.jpg'
    })
  });

  const result = await response.json();
  console.log(result.data); // { id, name, logo, city, created_at }

          import AddTeam from '@/components/AddTeam';

Problem: Import script fails
Solution: Make sure:
          1. npm run dev:safe is running
          2. teams.json exists
          3. JSON is valid array format
